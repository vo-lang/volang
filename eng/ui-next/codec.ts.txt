const textEncoder = new TextEncoder();
// Web text uses the platform UTF-8 replacement behavior for invalid guest bytes.
const textDecoder = new TextDecoder('utf-8', { ignoreBOM: true });
const nilArray = 0xffffffff;

class Writer {
  private data = new Uint8Array(256);
  private view = new DataView(this.data.buffer);
  private pos = 0;
  constructor(kind: number) { this.byte(86); this.byte(85); this.byte(73); this.byte(WIRE_VERSION); this.byte(kind); }
  private room(size: number): void {
    if (size > MAX_FRAME_BYTES - this.pos) throw new Error('UI frame exceeds limit');
    if (size <= this.data.length - this.pos) return;
    const grown = new Uint8Array(Math.min(MAX_FRAME_BYTES, Math.max(this.data.length * 2, this.pos + size)));
    grown.set(this.data.subarray(0, this.pos)); this.data = grown; this.view = new DataView(grown.buffer);
  }
  byte(value: number): void { this.room(1); this.data[this.pos++] = value; }
  length(value: number): void { this.room(4); this.view.setUint32(this.pos, value, true); this.pos += 4; }
  integer(value: number): void {
    if (!Number.isSafeInteger(value)) throw new Error('invalid UI integer');
    this.room(8); this.view.setUint32(this.pos, value >>> 0, true);
    this.view.setInt32(this.pos + 4, Math.floor(value / 0x100000000), true); this.pos += 8;
  }
  boolean(value: boolean): void { if (typeof value !== 'boolean') throw new Error('invalid UI boolean'); this.byte(value ? 1 : 0); }
  float(value: number): void {
    if (!Number.isFinite(value)) throw new Error('invalid UI float');
    this.room(8); this.view.setFloat64(this.pos, value, true); this.pos += 8;
  }
  string(value: string): void {
    if (typeof value !== 'string') throw new Error('invalid UI string');
    if (value.length > MAX_FRAME_BYTES - this.pos - 4) throw new Error('UI frame exceeds limit');
    const bytes = textEncoder.encode(value); this.length(bytes.length); this.room(bytes.length);
    this.data.set(bytes, this.pos); this.pos += bytes.length;
  }
  operation(value: Operation): void {
    const index = operations.indexOf(value);
    if (index < 0) throw new Error('invalid UI operation');
    this.byte(index + 1);
  }
  finish(): Uint8Array { return this.data.slice(0, this.pos); }
}

class Reader {
  private readonly view: DataView;
  private pos = 0;
  constructor(private readonly data: Uint8Array, kind: number) {
    if (!(data instanceof Uint8Array) || data.byteLength > MAX_FRAME_BYTES) throw new Error('UI frame exceeds limit');
    this.view = new DataView(data.buffer, data.byteOffset, data.byteLength);
    if (this.byte() !== 86 || this.byte() !== 85 || this.byte() !== 73 || this.byte() !== WIRE_VERSION || this.byte() !== kind) throw new Error('invalid UI frame header');
  }
  private need(size: number): void { if (size > this.data.length - this.pos) throw new Error('truncated UI frame'); }
  byte(): number { this.need(1); return this.data[this.pos++]; }
  length(): number { this.need(4); const value = this.view.getUint32(this.pos, true); this.pos += 4; return value; }
  integer(): number {
    this.need(8); const value = this.view.getUint32(this.pos, true) + this.view.getInt32(this.pos + 4, true) * 0x100000000;
    this.pos += 8; if (!Number.isSafeInteger(value)) throw new Error('invalid UI integer'); return value;
  }
  boolean(): boolean { const value = this.byte(); if (value > 1) throw new Error('invalid UI boolean'); return value === 1; }
  float(): number {
    this.need(8); const value = this.view.getFloat64(this.pos, true); this.pos += 8;
    if (!Number.isFinite(value)) throw new Error('invalid UI float'); return value;
  }
  string(): string { const size = this.length(); this.need(size); const bytes = this.data.subarray(this.pos, this.pos + size); this.pos += size; return textDecoder.decode(bytes); }
  operation(): Operation { const value = operations[this.byte() - 1]; if (!value) throw new Error('invalid UI operation'); return value; }
  count(minimum: number, limit: number): number {
    const count = this.length(); if (count === nilArray) return -1;
    if (count > limit || count > Math.floor((this.data.length - this.pos) / minimum)) throw new Error('invalid UI array count');
    return count;
  }
  finish(): void { if (this.pos !== this.data.length) throw new Error('trailing UI frame bytes'); }
}
